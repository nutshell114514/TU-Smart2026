```python
def approach_tune_log_update(now_ms):
    global _atlog_active, _atlog_t0, _atlog_last_ms, _atlog_wait_t0
    global _atlog_prev_mode, _atlog_prev_sub, _atlog_prev_ready
    global _atlog_prev_pmode, _atlog_prev_psub, _atlog_prev_pready
    global _atlog_prev_seen, _atlog_prev_plan, _atlog_prev_pause
    if not debug_switch.APPROACH_TUNE_LOG_ENABLE:
        return

    mode = _task_mode
    sub = _self_sub if mode == _MODE_APPROACH else _mode_sub
    pmode = base._Other_Car_Mode
    psub = base._Other_Car_Push_Sub
    pready = base._Other_Car_Ready
    if not _atlog_active:
        if mode != _MODE_APPROACH and pmode != _MODE_APPROACH:
            return
        _atlog_active = True
        _atlog_t0 = now_ms
        _atlog_last_ms = ticks_add(now_ms, -debug_switch.APPROACH_TUNE_LOG_PERIOD_MS)
        _atlog_wait_t0 = 0
        _atlog_prev_mode = mode
        _atlog_prev_sub = sub
        _atlog_prev_ready = _self_ready
        _atlog_prev_pmode = pmode
        _atlog_prev_psub = psub
        _atlog_prev_pready = pready
        _atlog_prev_seen = _ap_diag_seen
        _atlog_prev_plan = _approach_plan_valid
        _atlog_prev_pause = _ap_diag_pause_reason
        _approach_tune_log_cfg()
        print('AEV,0,START,mode=%d,sub=%d,pmode=%d,psub=%d,obj=%d,edge=%d' % (
            mode, sub, pmode, psub, _target_obj_id, _target_edge
        ))

    t = ticks_diff(now_ms, _atlog_t0)
    if mode != _atlog_prev_mode or sub != _atlog_prev_sub:
        print('AEV,%d,STATE,%d,%d,%d' % (t, mode, sub, _self_ready))
        _atlog_prev_mode = mode
        _atlog_prev_sub = sub
    if _self_ready != _atlog_prev_ready:
        print('AEV,%d,READY,%d' % (t, _self_ready))
        _atlog_prev_ready = _self_ready
    if pmode != _atlog_prev_pmode or psub != _atlog_prev_psub or pready != _atlog_prev_pready:
        print('AEV,%d,PEER_STATE,%d,%d,%d' % (t, pmode, psub, pready))
        _atlog_prev_pmode = pmode
        _atlog_prev_psub = psub
        _atlog_prev_pready = pready
    if _ap_diag_seen != _atlog_prev_seen:
        print('AEV,%d,SEEN,%d' % (t, _ap_diag_seen))
        _atlog_prev_seen = _ap_diag_seen
    if _approach_plan_valid != _atlog_prev_plan:
        print('AEV,%d,PLAN,%d,tyaw=%.1f,spin=%d,cmd_seq=%d' % (
            t, _approach_plan_valid, _approach_self_yaw, _approach_self_dir, _cmd_seq

        _atlog_prev_plan = _approach_plan_valid
    if _ap_diag_pause_reason != _atlog_prev_pause:
        print('AEV,%d,PAUSE,%d' % (t, _ap_diag_pause_reason))
        _atlog_prev_pause = _ap_diag_pause_reason

    if mode == _MODE_WAIT_READY and pmode == _MODE_WAIT_READY and pready >= 1:
        if _atlog_wait_t0 == 0:
            _atlog_wait_t0 = now_ms
        elif ticks_diff(now_ms, _atlog_wait_t0) >= 500:
            print('AEND,%d,BOTH_WAIT_READY' % t)
```

```python
# 功能：把世界坐标投影到“沿推送方向前进”的进度轴上。
```

## B1  里程计比例系数标定模式（_ODO_CALIB_* / _odo_calib_* / _odo_calib_step）

原属：`base.py`

```python
# ── 里程计比例系数标定（debug_switch.ODO_CALIB_MODE != 0 时生效）────────────
# 只做一件事：让车沿一个已知方向走一段，把里程计认为的位移显示出来，
# 和卷尺量到的实际距离相除就是系数的修正比例。
# 走的过程用世界速度闭环并锁死车头角，保证是纯平移，不掺转动。
_ODO_CALIB_WAIT_MS = const(4000)

_ODO_CALIB_MS = const(4000)

_ODO_CALIB_SPEED = const(40)

_odo_calib_t0 = 0

_odo_calib_yaw = 0.0

_odo_calib_x0 = 0.0

_odo_calib_y0 = 0.0

# 功能：里程计标定模式的每周期更新。
# 输入参数：now_ms 为当前时间戳。
def _odo_calib_step(now_ms):
    global _odo_calib_t0, _odo_calib_yaw, _odo_calib_x0, _odo_calib_y0
    if _odo_calib_t0 == 0:
        _odo_calib_t0 = now_ms
        _odo_calib_yaw = _car.current_angle
        _odo_calib_x0 = _car.Position_X
        _odo_calib_y0 = _car.Position_Y
    el = ticks_diff(now_ms, _odo_calib_t0)
    yr = radians(_odo_calib_yaw)
    c = cos(yr)
    s2 = sin(yr)
    if el < _ODO_CALIB_WAIT_MS:
        # 静止段：持续刷新起点，这样摆车、对齐标记期间的挪动不会算进结果。
        _odo_calib_x0 = _car.Position_X
        _odo_calib_y0 = _car.Position_Y
        request_hold(_odo_calib_yaw)
        st = 'WAIT %d' % ((_ODO_CALIB_WAIT_MS - el) // 100)
    elif el < _ODO_CALIB_WAIT_MS + _ODO_CALIB_MS:
        if debug_switch.ODO_CALIB_MODE == 1:
            request_world(s2 * _ODO_CALIB_SPEED, c * _ODO_CALIB_SPEED, _odo_calib_yaw)
        else:
            request_world(c * _ODO_CALIB_SPEED, -s2 * _ODO_CALIB_SPEED, _odo_calib_yaw)
        st = 'RUN %d' % ((_ODO_CALIB_WAIT_MS + _ODO_CALIB_MS - el) // 100)
    else:
        request_hold(_odo_calib_yaw)
        st = 'DONE'
    dx = _car.Position_X - _odo_calib_x0
    dy = _car.Position_Y - _odo_calib_y0
    # 投影回起始车头方向：fwd 沿车头，lat 沿车体右方。
    fwd = s2 * dx + c * dy
    lat = c * dx - s2 * dy
    if _lcd is not None:
        pad = '                      '
        rows = ((0, 'ODO CALIB m%d %s' % (debug_switch.ODO_CALIB_MODE, st)),
                (16, 'fwd %.2f' % fwd),
                (32, 'lat %.2f' % lat),
                (48, 'KX %.7f' % _MILEAGE_KX),
                (64, 'KY %.7f' % _MILEAGE_KY),
                (80, 'yaw0 %.1f now %.1f' % (_odo_calib_yaw, _car.current_angle)),
                (100, 'dx %.1f dy %.1f' % (dx, dy)))
        for y, t in rows:
            _lcd.str12(0, y, (t + pad)[:_LCD_MAX_CHARS])
```

## B2  cone 固定锚点修坐标（config._CONE_WORLD_POINTS 两个锚点均为 999，永远匹配不上）

原属：`base.py`

```python
# 功能：主车使用配置中的固定 cone 世界坐标反算并修正自身 X/Y。
# 先用当前主车位姿计算出的 _cam_cone_x/y 识别锚点；X/Y 都在 ±40cm 门内才匹配。
# 匹配不到立即忽略。匹配成功后通过外部位置修正请求同时更新车辆位置和编码器基准，
# 并把本帧 cone 世界坐标改为锚点真值，供避障表和无线发送继续使用。
def cone_anchor_position_fix(c_yaw, s_yaw):
    global _cam_cone_x, _cam_cone_y
    global _pos_fix_req, _pos_fix_x, _pos_fix_y, _pos_fix_x_valid, _pos_fix_y_valid
    if not config._IS_LEADER:
        return 0
    if not _point_valid(_cam_cone_x, _cam_cone_y):
        return 0
    gate = _CONE_ANCHOR_MATCH_GATE_CM
    best_x = 999.0
    best_y = 999.0
    best_err = 99999999.0
    for anchor_x, anchor_y in config._CONE_WORLD_POINTS:
        if not _point_valid(anchor_x, anchor_y):
            continue
        dx = _cam_cone_x - anchor_x
        dy = _cam_cone_y - anchor_y
        if abs(dx) > gate or abs(dy) > gate:
            continue
        err = dx * dx + dy * dy
        if err < best_err:
            best_err = err
            best_x = anchor_x
            best_y = anchor_y
    if not _point_valid(best_x, best_y):
        return 0
    new_car_x = best_x - c_yaw * _cam_cone_rel_x - s_yaw * _cam_cone_rel_y
    new_car_y = best_y + s_yaw * _cam_cone_rel_x - c_yaw * _cam_cone_rel_y
    _pos_fix_x = new_car_x
    _pos_fix_y = new_car_y
    _pos_fix_x_valid = True
    _pos_fix_y_valid = True
    _pos_fix_req = True
    _cam_cone_x = best_x
    _cam_cone_y = best_y
    return 1
```

```python
# 功能：拼出一行"球实测坐标 + 该侧紧 V 目标参考"的显示串，供手动摆 V 形时直接读数回填
# task._PUSH_BALL_FIXED_REF_BY_OBJ。
# 输入参数：tag 为槽位标记 'L'/'R'；slot 为球槽位下标（0=紫球/左槽位，1=橙球/右槽位）。
# 返回值：形如 "L 8.2,16.4 r8,17" 的字符串；球无效时实测部分显示 --。
def _ball_ref_line(tag, slot):
    obj_id = task._target_obj_id
    idx = ((obj_id - 1) if 1 <= obj_id <= _OBJ_ID_MAX else 0) * 4 + slot * 2
    ref_x = task._PUSH_BALL_FIXED_REF_BY_OBJ[idx]
    ref_y = task._PUSH_BALL_FIXED_REF_BY_OBJ[idx + 1]
    mx = _cam_ball_rel_x[slot]
    my = _cam_ball_rel_y[slot]
    if mx >= 900.0 or my >= 900.0:
        return '%s --,-- r%.0f,%.0f' % (tag, ref_x, ref_y)
    return '%s %.1f,%.1f r%.0f,%.0f' % (tag, mx, my, ref_x, ref_y)
```

```python
            # 无线解析目标：由球参考 + 主车球安装位 bL 反算出"球正好落在参考点时，
            # 主车中心该在从车车体系的哪里"。它和球参考同源，收敛到它就必然重新看得见球。
            # 原来用的是起推那一刻锁存的 _wl_ref —— 那只是"当时碰巧什么样"，
            # 如果起推已经有偏差，无线就把偏差当目标守着，从车永远爬不回球的视野。
            an_ok, an_x, an_y = _wl_ref_analytic(now_ms)
            if an_ok:
                # 无线通路单独加硬：把误差按 _PUSH_WL_GAIN 放大后再交给球层 PID。
                # 等效于只在无线通路上抬高 Kp/Kd/Ki，球通路的手感完全不受影响，
                # 也不用把参数一路穿到 base.request_push 里去。
                if _PUSH_WL_GAIN != 1.0:
                    wl_x = an_x + (wl_x - an_x) * _PUSH_WL_GAIN
                    wl_y = an_y + (wl_y - an_y) * _PUSH_WL_GAIN
                return (True, wl_x, wl_y, True, an_x, an_y)
            if _wl_ref_valid:
                return (True, wl_x, wl_y, True, _wl_ref_x, _wl_ref_y)
        return (False, 0.0, 0.0, _armed, _ref_x, _ref_y)
    if sel_id == _BALL_L_CLASS_ID:
        x, y = (x_l, y_l)
    else:
        x, y = (x_r, y_r)
    if sel_id != _ref_id:
        if _PUSH_BALL_DYNAMIC_REF_ENABLE:
            if _off_valid:
                if sel_id == _BALL_R_CLASS_ID:
                    _ref_x += _off_x
                    _ref_y += _off_y
                else:
                    _ref_x -= _off_x
                    _ref_y -= _off_y
            else:
                _ref_x = x
                _ref_y = y
        else:
            idx = (_target_obj_id - 1) * 4
            if sel_id == _BALL_L_CLASS_ID:
                _ref_x = _PUSH_BALL_FIXED_REF_BY_OBJ[idx]
                _ref_y = _PUSH_BALL_FIXED_REF_BY_OBJ[idx + 1]
            else:
                _ref_x = _PUSH_BALL_FIXED_REF_BY_OBJ[idx + 2]
                _ref_y = _PUSH_BALL_FIXED_REF_BY_OBJ[idx + 3]
        _ref_id = sel_id
        _prev_x = x
        _prev_y = y
        _prev_valid = True
        _jump_cnt = 0
        return (True, x, y, True, _ref_x, _ref_y)
    if _prev_valid:
        dx = x - _prev_x
        dy = y - _prev_y
        if dx * dx + dy * dy > _BALL_JUMP_MAX_CM * _BALL_JUMP_MAX_CM:
            _jump_cnt += 1
            if _jump_cnt < 3:
                return (False, 0.0, 0.0, True, _ref_x, _ref_y)
    if _wl_ref_valid:
        wl_ok, wl_x, wl_y = _wl_rel()
        if wl_ok:
            ex = x - _ref_x - (wl_x - _wl_ref_x)
            ey = y - _ref_y - (wl_y - _wl_ref_y)
            if ex * ex + ey * ey > _BALL_WL_XCHECK_CM * _BALL_WL_XCHECK_CM:
                return (True, wl_x, wl_y, True, _wl_ref_x, _wl_ref_y)
    _jump_cnt = 0
    _prev_x = x
    _prev_y = y
    _prev_valid = True
    return (True, x, y, True, _ref_x, _ref_y)
```

```python
    push_rad = radians(pyaw)
    if _push_req_ball_seen and _push_req_ball_ref_valid and (_push_ball_seen_seq != _push_req_seq):
        # 球层：先把车体系误差旋转到"推送轴 / V 张开轴"再做 PD。
        # 从车车头角 lock_yaw = push_yaw − 45°，所以车体 x/y 轴各自都是
        # "沿推送方向落后"和"V 张开"的 45° 混合（纯落后 3cm → err=(2.12,2.12)，
        # 纯分离 3cm → err=(−2.12,2.12)）。直接在车体轴上调增益，改一个必然动另一个。
        # 旋转解耦后两个物理通道彻底分开：推送轴权限管"跟不上"，张开轴权限管"漏球"。
        _push_ball_seen_seq = _push_req_seq
        _push_ball_lost_t0 = ticks_ms()
        err_x = _push_req_ball_rel_x - _push_req_ball_ref_x
        err_y = _push_req_ball_rel_y - _push_req_ball_ref_y
        # 球【只修推送方向的垂直分量】，也就是 V 张开的那个轴。
        # 沿推送方向的位置由各车自己的物体环负责，球不插手——车体轴和推送轴差 45°，
        # 如果球在车体轴上工作，它会同时干预推送方向，和物体环打架。
        d_rad = radians((pyaw - _pid__push_lock_yaw + 180.0) % 360.0 - 180.0)
        sd = sin(d_rad)
        cd = cos(d_rad)
        e_open = -err_x * cd + err_y * sd
        _push_ball_dy_f = _push_ball_dy_f * (1.0 - _p_push_d_lpf) + (e_open - _push_ball_err_y_prev) * _p_push_d_lpf
        _push_ball_err_y_prev = e_open
        # e_open 的符号是【跟槽位走的】：分离在一侧为正、另一侧为负（命令映射会自动
        # 抵消这个翻转，方向始终正确），但不对称判据必须用与槽位无关的量，否则会在
        # 其中一个槽位把"收紧"当成"过紧"削弱掉。sd 的符号正好标识槽位。
        e_sep = e_open if sd >= 0.0 else -e_open
        # e_sep > 0 = 两车分离（物体要从 V 中间漏出去）→ 全权限强力收紧；
        # e_sep < 0 = 靠得过紧 → 增益和限幅乘 relax。物体漏不出去，宁可过紧。
        open_kp = _p_push_ball_side_kp
        open_max = _p_push_ball_side_max
        if e_sep < 0.0:
            open_kp = open_kp * _p_push_ball_open_relax
            open_max = open_max * _p_push_ball_open_relax
        open_raw = open_kp * _mot_deadband(e_open, _p_push_ball_deadband) + _p_push_ball_side_kd * _push_ball_dy_f
        open_raw = _mot_limit(open_raw, open_max)
        _push_ball_ch_open = _mot_slew(open_raw, _push_ball_ch_open, _p_push_ball_slew)
        # 纯垂直命令旋转回车体系。
        _push_ball_side_cmd = -_push_ball_ch_open * cd
        _push_ball_fwd_cmd = _push_ball_ch_open * sd
    if _push_ball_lost_t0 and ticks_diff(ticks_ms(), _push_ball_lost_t0) > _p_push_ball_lost_tol_ms:
        _push_ball_side_cmd = 0.0
        _push_ball_fwd_cmd = 0.0

# 球层命令叠加到世界速度：
    vy += -s * _push_side_cmd + cos(fwd_push_rad) * _push_fwd_cmd * fwd_sign
    vx += c * _push_ball_side_cmd + s * _push_ball_fwd_cmd
```
